
CLASS Pet:
    # Constructor to initialize common pet attributes
    FUNCTION __init__(name, age, breed):
        SET this.name = name
        SET this.age = age
        SET this.breed = breed
    ENDFUNCTION

    # A general method for displaying info
    FUNCTION display_info():
        PRINT "Name:", this.name
        PRINT "Age:", this.age
        PRINT "Breed:", this.breed
    ENDFUNCTION
CLASS Dog EXTENDS Pet:
    # Constructor for Dog
    FUNCTION __init__(name, age, breed, size):
        # Call the parent class constructor to initialize common attributes
        CALL SUPER.__init__(name, age, breed)
        # Initialize dog-specific attribute
        SET this.size = size
    ENDFUNCTION
    FUNCTION make_sound():
        PRINT "Woof!"
    ENDFUNCTION
ENDCLASS
